        async void gridView1_ValidateRow(object sender, DevExpress.XtraGrid.Views.Base.ValidateRowEventArgs e)
        { // https://docs.devexpress.com/WindowsForms/DevExpress.XtraGrid.Views.Base.ColumnView.ValidateRow
             {entity} x = ({entity}) e.Row;

             //}

             ////////////////////////////////////////////database logic

                 x.{PKL} = await _{entityL}Service.InsertReturnIdAsync(x);
             else

                 }
             }
